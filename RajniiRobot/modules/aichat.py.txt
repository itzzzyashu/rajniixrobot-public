            )

async def get_message(context: CallbackContext, message):
    reply_message = message.reply_to_message
    if message.text.lower() in ["rajni", "rajnii", "rajnibot", "rajniixbot", "rajniixrobot", "rajnixbot", "rajnibahu", ""]:
        return True
    if reply_message:
        if reply_message.from_user.id == context.bot.get_me().id:
            return True
    else:
        return False


async def ai_reply(update: Update, context: CallbackContext):
    message = update.effective_message
    chat_id = update.effective_chat.id
    bot = context.bot
    is_aichatbot_enabled = sql.is_aichatbot_enabled(chat_id)
    if not is_aichatbot_enabled:
        return

    if message.text and not message.document:
        if not get_message(context, message):
            return
        Message = message.text
        bot.send_chat_action(chat_id, action="typing")
        response = await requests.get(
            AI_API_URL,
            params={ "user_id": chat_id, "query": Message, "BOT_ID": BOT_ID },
            headers={ "api_key": AI_API_KEY })
        data = await response.json()
        if data['status'] == 200:
            reply = data['result']['text']
            message.reply_text(reply, timeout=60)
        else:
            await RAJNI_PTB.send_message(
                SUPPORT_CHAT,
                text=f"*➢ ᴄʜᴀᴛʙᴏᴛ ᴇʀʀᴏʀ:*\n`{data['msg']}`\n\n➢ ʀᴇᴘᴏʀᴛᴇᴅ ꜰʀᴏᴍ ᴄʜᴀᴛ ([{context.bot.get_chat(chat_id).title}](https://telegram.me/c/{abs(chat_id)}) | `{chat_id}`)",
                reply_markup=InlineKeyboardMarkup([
                    [
                        InlineKeyboardButton(
                            text="► ᴘᴏᴡᴇʀᴇᴅ ʙʏ ᴅᴇᴠꜱʟᴀʙ ◄",
                            url="https://telegram.me/devslab"),
                    ]
                ])
            )
            message.reply_text(
                f"*➢ ᴄʜᴀᴛʙᴏᴛ ᴇʀʀᴏʀ:* `{data['msg']}`\n\n➢ ʀᴇᴘᴏʀᴛᴇᴅ ᴛᴏ ᴛʜᴇ @{SUPPORT_CHAT}.",
                parse_maode=ParseMode.MARKDOWN,
                timeout=60,
                reply_markup=InlineKeyboardMarkup([
                    [
                        InlineKeyboardButton(
                            text="► ꜱᴜᴘᴘᴏʀᴛ ◄",
                            url=f"https://telegram.me/{SUPPORT_CHAT}"),
                        InlineKeyboardButton(
                            text="► ᴜᴘᴅᴀᴛᴇꜱ ◄",
                            url=f"https://telegram.me/{UPDATES_LOGS}")
                    ],
                    [
                        InlineKeyboardButton("► ᴘᴏᴡᴇʀᴇᴅ ʙʏ ᴅᴇᴠꜱʟᴀʙ ◄", url="https://telegram.me/devslab"),
                    ]
                ])
            )



def ai_chats(update: Update, context: CallbackContext):
    chats = sql.get_all_aichatbot_chats()
    text = "<b>ᴀɪ ᴄʜᴀᴛʙᴏᴛ ɪꜱ ᴇɴᴀʙʟᴇᴅ ɪɴ ᴛʜᴇꜱᴇ ᴄʜᴀᴛꜱ</b>\n"
    for chat in chats:
        try:
            x = context.bot.get_chat(int(*chat))
            name = x.title or x.first_name
            text += f"➢ <code>{name}</code>\n"
        except (BadRequest, TelegramError):
            sql.rem_aichatbot(*chat)
        except RetryAfter as e:
            sleep(e.retry_after)
    update.effective_message.reply_text(text, parse_mode=ParseMode.HTML)
